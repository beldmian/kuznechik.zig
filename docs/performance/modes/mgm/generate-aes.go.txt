package main
import("crypto/aes"; "crypto/sha256"; "fmt"; "independent/mgm")
func main(){
key:=make([]byte,16); nonce:=make([]byte,16); aad:=make([]byte,1057); input:=make([]byte,2081)
for i:=range key {key[i]=byte(7+11*i); nonce[i]=byte(13+7*i)}
for i:=range aad {aad[i]=byte(3+17*i)}
for i:=range input {input[i]=byte(5+19*i)}
c,_:=aes.NewCipher(key); m,_:=mgm.NewMGM(c); hash:=sha256.New(); count:=0
run:=func(a,n int){if a+n==0{return}; out:=m.Seal(nil,nonce,input[:n],aad[:a]); hash.Write(out);count++}
for a:=0;a<17;a++ {for n:=0;n<33;n++ {run(a,n)}}
lengths:=[]int{0,1,15,16,17,1007,1008,1009,1023,1024,1025,1039,1040,1041,1057}
for _,a:=range lengths{for _,n:=range append(append([]int{},lengths...),2047,2048,2049,2081){run(a,n)}}
fmt.Printf("cases=%d sha256=%x\n",count,hash.Sum(nil))
}
